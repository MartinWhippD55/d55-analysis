- **What:** Implement Glue catalog client: AssumeRole -> Project Role creds, list databases/tables, filter to bryt_number tables, return AvailableDataSource[] with columns
- **Why:** This is the core of `shared-lib:glue-catalog-client` and the read side of discovery. It unblocks the Data Source API (US-03) `list-available` handler and the render enrichment step (US-05).
- **Done when:**
  - The module assumes the Project Role (`PROJECT_ROLE_ARN`) via `sts:AssumeRole` and uses the temporary credentials for all Glue calls.
  - `listAvailableDataSources()` lists databases/tables in the project's Glue catalog and returns `AvailableDataSource[]`, each with `database`, `tableName`, and `columns`.
  - Only tables containing a `bryt_number` column are returned; tables without it are filtered out.
  - AssumeRole failure and an unreachable Glue catalog are logged and propagated so the caller can return 500/503.

### Suggested approach

```typescript
import { STSClient, AssumeRoleCommand } from "@aws-sdk/client-sts";
import { GlueClient, GetDatabasesCommand, GetTablesCommand } from "@aws-sdk/client-glue";
import type { AvailableDataSource, DataSourceColumn } from "shared-lib/data-source-types";

const BRYT_JOIN_COLUMN = "bryt_number";

async function assumeProjectRole(): Promise<GlueClient> {
  const sts = new STSClient({});
  const { Credentials } = await sts.send(
    new AssumeRoleCommand({
      RoleArn: process.env.PROJECT_ROLE_ARN!,
      RoleSessionName: "glue-catalog-discovery",
    }),
  );
  if (!Credentials) throw new Error("AssumeRole returned no credentials");
  return new GlueClient({
    credentials: {
      accessKeyId: Credentials.AccessKeyId!,
      secretAccessKey: Credentials.SecretAccessKey!,
      sessionToken: Credentials.SessionToken!,
    },
  });
}

export async function listAvailableDataSources(): Promise<AvailableDataSource[]> {
  const glue = await assumeProjectRole();
  const sources: AvailableDataSource[] = [];

  const { DatabaseList = [] } = await glue.send(new GetDatabasesCommand({}));
  for (const db of DatabaseList) {
    const { TableList = [] } = await glue.send(new GetTablesCommand({ DatabaseName: db.Name }));
    for (const table of TableList) {
      const columns: DataSourceColumn[] = (table.StorageDescriptor?.Columns ?? []).map((c) => ({
        name: c.Name!,
        type: c.Type!,
      }));
      // Only bryt_number-joinable tables are discoverable.
      if (columns.some((c) => c.name === BRYT_JOIN_COLUMN)) {
        sources.push({
          database: db.Name!,
          tableName: table.Name!,
          columns,
          location: table.StorageDescriptor?.Location,
        });
      }
    }
  }
  return sources;
}
```
