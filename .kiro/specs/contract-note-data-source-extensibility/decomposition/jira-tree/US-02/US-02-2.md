- **What:** Implement column detail fetcher: full column list (name, type) for a specific {database}/{table}
- **Why:** The Section Editor field browser and shared-section dependency checks need the full column detail for one specific table. It unblocks the Data Source API (US-03) `get-columns` handler.
- **Done when:**
  - `getDataSourceColumns(database, table)` returns `DataSourceColumn[]` for the given `{database}/{table}`, each column carrying `name` and its Glue/Athena `type`.
  - A table not present in the catalog propagates a not-found signal so the caller can return 404.
  - The fetcher reuses the assumed Project Role credentials (same path as discovery).

### Suggested approach

```typescript
import { GlueClient, GetTableCommand, EntityNotFoundException } from "@aws-sdk/client-glue";
import type { DataSourceColumn } from "shared-lib/data-source-types";

export async function getDataSourceColumns(
  database: string,
  table: string,
): Promise<DataSourceColumn[]> {
  const glue = await assumeProjectRole(); // same AssumeRole path as discovery
  try {
    const { Table } = await glue.send(
      new GetTableCommand({ DatabaseName: database, Name: table }),
    );
    return (Table?.StorageDescriptor?.Columns ?? []).map((c) => ({
      name: c.Name!,
      type: c.Type!, // Glue/Athena type: string, int, bigint, double, boolean, ...
    }));
  } catch (err) {
    if (err instanceof EntityNotFoundException) {
      throw new Error(`Table not found: ${database}/${table}`); // caller -> 404
    }
    throw err;
  }
}
```
