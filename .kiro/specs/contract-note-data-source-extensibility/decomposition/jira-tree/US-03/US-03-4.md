- **What:** Implement attach-data-source handler (validate table exists + has bryt_number; write DATASOURCE record)
- **Why:** Backs `POST /contract-note-templates/{templateId}/data-sources`, the write path the frontend uses to associate a data source with a template. This is one side of Property 2 (attachment round-trip). Depends on the Glue client (US-02) for validation and the shared types (US-01) for the record shape.
- **Done when:**
  - `data-sources/attach-data-source.ts` validates via the Glue client that the table exists and has a `bryt_number` column.
  - A missing table returns 404; a table without `bryt_number` returns 400 with a validation message.
  - On success it writes a `DATASOURCE#{database}#{tableName}` record under `TEMPLATE#{templateId}` with `entityType: "TemplateDataSource"`, `displayName` (defaulting to the table name), `attachedAt` (ISO 8601), and `attachedBy` (Cognito username).
  - Attachment is allowed regardless of template DRAFT/PUBLISHED status.

### Suggested approach

```typescript
// api/src/data-sources/attach-data-source.ts
export const handler = async (event: APIGatewayProxyEvent): Promise<APIGatewayProxyResult> => {
  const { templateId } = event.pathParameters ?? {};
  const { database, tableName, displayName } = JSON.parse(event.body ?? '{}');

  const glue = await getGlueCatalogClient(process.env.PROJECT_ROLE_ARN!);
  const t = await glue.getTable(database, tableName);
  if (!t) return { statusCode: 404, body: JSON.stringify({ message: 'Table not found' }) };
  if (!t.columns.some((c) => c.name === 'bryt_number'))
    return { statusCode: 400, body: JSON.stringify({ message: 'Table missing bryt_number column' }) };

  const record: TemplateDataSourceRecord = {
    PK: `TEMPLATE#${templateId}`,
    SK: `DATASOURCE#${database}#${tableName}`,
    entityType: 'TemplateDataSource',
    database, tableName,
    displayName: displayName ?? tableName,
    attachedAt: new Date().toISOString(),
    attachedBy: getUsername(event),
  };
  await ddb.put({ TableName: TABLE, Item: record });

  return { statusCode: 201, body: JSON.stringify(toTemplateDataSource(record)) };
};
// Request: { "database": "energy_prod", "tableName": "contracts", "displayName": "Contracts" }
```
