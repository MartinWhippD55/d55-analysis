- **What:** Implement detach-data-source handler (scan all sections' variants for referencing fields; 409 with affected section+variant list, else remove)
- **Why:** Backs `DELETE /contract-note-templates/{templateId}/data-sources/{database}/{table}` and implements Property 3 (detachment with variant-field-in-use warning). Prevents silently breaking a template whose section variants still reference the data source.
- **Done when:**
  - `data-sources/detach-data-source.ts` scans every section's variants in the template for fields referencing `{database}/{table}` (the namespaced field-reference format US-04's scanner also uses).
  - If any variant references it, the handler returns **409** with the affected section+variant list and does **not** delete the record.
  - If unreferenced, it deletes the `DATASOURCE#{database}#{tableName}` record under `TEMPLATE#{templateId}`.
  - Detach is allowed regardless of template DRAFT/PUBLISHED status.

### Suggested approach

```typescript
// api/src/data-sources/detach-data-source.ts
export const handler = async (event: APIGatewayProxyEvent): Promise<APIGatewayProxyResult> => {
  const { templateId, database, table } = event.pathParameters ?? {};

  const sections = await loadTemplateSections(templateId!);
  const affected: { sectionId: string; variantId: string }[] = [];
  for (const section of sections) {
    for (const variant of section.variants) {
      if (variantReferencesDataSource(variant, database!, table!)) {
        affected.push({ sectionId: section.id, variantId: variant.id });
      }
    }
  }

  if (affected.length > 0) {
    return {
      statusCode: 409,
      body: JSON.stringify({ message: 'Data source is in use by section variants', affected }),
    };
  }

  await ddb.delete({
    TableName: TABLE,
    Key: { PK: `TEMPLATE#${templateId}`, SK: `DATASOURCE#${database}#${table}` },
  });
  return { statusCode: 204, body: '' };
};
// 409 body: { "message": "...", "affected": [{ "sectionId": "sec-1", "variantId": "var-a" }] }
```
