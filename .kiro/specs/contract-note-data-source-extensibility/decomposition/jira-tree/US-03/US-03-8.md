- **What:** Create DataSourceApi CDK construct (per-op NodejsFunctions, grant table/Glue/Athena/AssumeRole, wire LambdaIntegrations, pass PROJECT_ROLE_ARN + Athena config); instantiate in contract-note-stack.ts
- **Why:** The construct is what actually deploys the six handlers (US-03-2..7), grants them the IAM access they need, and binds them to the routes from US-03-1. Without it the handlers exist but are unreachable. Mirrors the existing `template-api.ts` convention (Requirement 7.6).
- **Done when:**
  - `DataSourceApi` creates one `NodejsFunction` per operation from `api/src/data-sources/` (list-available, get-columns, list-attached, attach-data-source, detach-data-source, list-shared-section-deps).
  - Each function is granted table read/write, Glue and Athena access, and `sts:AssumeRole` for the Project Role.
  - `list-available` and `get-columns` additionally receive `PROJECT_ROLE_ARN` and the Athena config as environment variables.
  - `LambdaIntegration`s are wired to the `DataSourceRouteResources` from `contract-note-foundation.ts`.
  - The construct is instantiated in `contract-note-stack.ts` and `cdk synth` succeeds.

### Suggested approach

```typescript
// cdk/lib/data-source-api.ts
export interface DataSourceApiProps {
  table: dynamodb.ITable;
  routes: DataSourceRouteResources;
  projectRoleArn: string;
  athenaWorkgroup: string;
  athenaResultsBucket: s3.IBucket;
}

export class DataSourceApi extends Construct {
  constructor(scope: Construct, id: string, props: DataSourceApiProps) {
    super(scope, id);

    const fn = (name: string, entry: string, extraEnv: Record<string, string> = {}) => {
      const f = new nodejs.NodejsFunction(this, name, {
        entry: `api/src/data-sources/${entry}`,
        environment: { TABLE_NAME: props.table.tableName, ...extraEnv },
      });
      props.table.grantReadWriteData(f);
      f.addToRolePolicy(new iam.PolicyStatement({ actions: ['sts:AssumeRole'], resources: [props.projectRoleArn] }));
      f.addToRolePolicy(new iam.PolicyStatement({ actions: ['glue:GetTable', 'glue:GetTables', 'glue:GetDatabases', 'athena:*'], resources: ['*'] }));
      return f;
    };

    const glueEnv = { PROJECT_ROLE_ARN: props.projectRoleArn, ATHENA_WORKGROUP: props.athenaWorkgroup };
    const listAvailable = fn('ListAvailable', 'list-available.ts', glueEnv);
    const getColumns    = fn('GetColumns', 'get-columns.ts', glueEnv);
    const listAttached  = fn('ListAttached', 'list-attached.ts');
    const attach        = fn('Attach', 'attach-data-source.ts', glueEnv);
    const detach        = fn('Detach', 'detach-data-source.ts');
    const sharedDeps    = fn('SharedSectionDeps', 'list-shared-section-deps.ts');

    props.routes.dataSources.addMethod('GET', new apigateway.LambdaIntegration(listAvailable));
    props.routes.dataSourceColumns.addMethod('GET', new apigateway.LambdaIntegration(getColumns));
    props.routes.templateDataSources.addMethod('GET', new apigateway.LambdaIntegration(listAttached));
    props.routes.templateDataSources.addMethod('POST', new apigateway.LambdaIntegration(attach));
    props.routes.templateDataSource.addMethod('DELETE', new apigateway.LambdaIntegration(detach));
    props.routes.sharedSectionDeps.addMethod('GET', new apigateway.LambdaIntegration(sharedDeps));
  }
}
```
