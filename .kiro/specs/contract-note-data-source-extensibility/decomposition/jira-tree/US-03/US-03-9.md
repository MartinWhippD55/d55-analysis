- **What:** Property tests for data source API (Property 2: attachment round-trip; Property 3: detachment with variant-field-in-use warning)
- **Why:** Locks in the two behaviours that example-based tests miss across the input space: attaching then listing always round-trips (Property 2), and detaching an in-use data source is always blocked with the affected list (Property 3). Optional — deferrable for a faster MVP.
- **Done when:**
  - A property test generates random valid attachments; attach-then-list always includes the attachment with matching database, tableName, and displayName (Property 2).
  - A property test generates random schemas where a variant references an attached data source; detach always returns 409 with the affected section+variant list (Property 3).
  - Both properties run under the project's hypothesis/fast-check runner and pass.

### Suggested approach

```python
# tests/property/test_data_source_api.py
from hypothesis import given, strategies as st

databases = st.text(min_size=1, max_size=20, alphabet=st.characters(whitelist_categories=("Ll", "Nd")))
tables = st.text(min_size=1, max_size=20, alphabet=st.characters(whitelist_categories=("Ll", "Nd")))

# Property 2: attachment round-trip
@given(database=databases, table_name=tables, display=st.text(min_size=1, max_size=40))
def test_attach_then_list_includes_attachment(database, table_name, display):
    template_id = new_template()
    attach(template_id, database, table_name, display)          # POST .../data-sources
    listed = list_attached(template_id)                          # GET  .../data-sources
    assert any(
        d["database"] == database and d["tableName"] == table_name and d["displayName"] == display
        for d in listed
    )

# Property 3: detachment with variant-field-in-use warning
@given(database=databases, table_name=tables, refs=st.lists(st.tuples(st.text(min_size=1), st.text(min_size=1)), min_size=1))
def test_detach_in_use_is_blocked_with_affected_list(database, table_name, refs):
    template_id = new_template()
    attach(template_id, database, table_name, table_name)
    for section_id, variant_id in refs:                          # variants reference the data source
        add_variant_field_ref(template_id, section_id, variant_id, database, table_name)

    resp = detach(template_id, database, table_name)             # DELETE .../data-sources/{db}/{table}
    assert resp.status_code == 409
    affected = resp.json()["affected"]
    for section_id, variant_id in refs:
        assert {"sectionId": section_id, "variantId": variant_id} in affected
```
