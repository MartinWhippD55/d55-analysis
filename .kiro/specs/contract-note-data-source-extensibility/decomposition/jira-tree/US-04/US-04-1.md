- **What:** Implement pdf-me schema field-reference scanner: walk all pages of { schemas: [[...]] }, collect element names containing '.', map prefix -> table name
- **Why:** The scanner is the core primitive of `shared-lib:dependency-scanner`; the recompute in US-04-2 and the downstream US-07 dependency check both rely on it to turn a schema into a set of data source tables.
- **Done when:**
  - Given schema JSON (`{ schemas: [[...]] }`), every page array is walked and each element `name` containing a `.` is collected.
  - Each namespaced name's prefix is mapped to a data source table name (`credit_data.credit_score` → `credit_data`).
  - Core (non-`.`) element names are ignored and repeated references are de-duplicated.

### Suggested approach

```typescript
// shared-lib/dependency-scanner.ts
// Walk a pdf-me Schema_JSON document and collect the data source tables it references.
// A data-source field is any element whose `name` is namespaced `{table}.{column}`.

interface PdfMeSchema {
  schemas: Array<Array<{ name?: string }>>; // pages -> elements
}

/** Return the distinct data source table names referenced by one schema document. */
export function scanSchemaForDataSources(schema: PdfMeSchema): Set<string> {
  const tables = new Set<string>();
  for (const page of schema.schemas ?? []) {
    for (const element of page ?? []) {
      const name = element?.name;
      if (!name || !name.includes(".")) continue; // core contract field — ignore
      const prefix = name.slice(0, name.indexOf("."));
      if (prefix) tables.add(prefix); // prefix -> data source table name
    }
  }
  return tables;
}
```
