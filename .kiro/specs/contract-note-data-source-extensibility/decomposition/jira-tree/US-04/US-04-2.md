- **What:** Implement shared-section dependency recompute: on variant schema save / version publish, scan all variants' schemas, compute union of referenced data sources, reconcile DATASOURCE_DEP records
- **Why:** This is the recompute half of `shared-lib:dependency-scanner`. It keeps a shared section's `DATASOURCE_DEP` records in sync with its variants so US-07's attachment-time dependency check has authoritative data to read.
- **Done when:**
  - On a shared section variant schema save or version publish, all of that section's variants' current schemas are scanned and the union of referenced data sources is computed.
  - `DATASOURCE_DEP` records are reconciled to equal that union — newly-referenced sources are added, sources no longer referenced by any variant are removed.
  - It hooks into the existing `save-section-schema` / `publish-section-version` flow for shared sections and fails loudly on a missing/unreadable variant schema rather than persisting a partial union.

### Suggested approach

```typescript
// shared-lib/dependency-scanner.ts
// Reconcile a shared section's DATASOURCE_DEP records to the union of data sources
// referenced across all of its variants' schemas. Called from save-section-schema /
// publish-section-version for shared sections.
import { scanSchemaForDataSources } from "./scan-schema";
import type { SectionDataSourceDependency } from "shared-lib/types"; // US-01

export async function recomputeSharedSectionDependencies(sharedSectionId: string) {
  const variants = await loadVariants(sharedSectionId);

  // Union across all variants — a section depends on any source any variant references.
  const desired = new Set<string>();
  for (const variant of variants) {
    const schema = await loadSchemaJson(variant); // throws if missing/unreadable
    for (const table of scanSchemaForDataSources(schema)) desired.add(table);
  }

  const existing = await listDatasourceDeps(sharedSectionId); // SK: DATASOURCE_DEP#{database}#{tableName}
  const existingTables = new Set(existing.map((d) => d.tableName));

  const toAdd = [...desired].filter((t) => !existingTables.has(t));
  const toRemove = existing.filter((d) => !desired.has(d.tableName));

  await Promise.all([
    ...toAdd.map((tableName) => putDatasourceDep(sharedSectionId, tableName)),
    ...toRemove.map((dep: SectionDataSourceDependency) => deleteDatasourceDep(dep)),
  ]);
}
```
