- **What:** Property tests for enrichment (Property 7: namespaced data; 8: empty pass-through; 9: empty rows continue; 10: failure routes to handle-failure)
- **Why:** Property tests pin the four correctness properties of the enrichment state (namespacing, pass-through, empty-rows continuation, failure routing) against generated inputs, catching edge cases example tests miss. Optional — deferrable for a faster MVP.
- **Done when:**
  - Property 7 holds: attached data sources + a valid BrytNumber yield columns under `{tableName}.{columnName}`.
  - Property 8 holds: no attachments returns `contractData` unchanged with zero Athena queries.
  - Property 9 holds: a zero-row query leaves those fields empty and rendering continues.
  - Property 10 holds: a failing query (Athena error/timeout) throws so the state machine routes to `handle-failure`.

### Suggested approach

```python
# Hypothesis sketch — generators produce random attachments + Athena results.
from hypothesis import given, strategies as st

table_names = st.text(min_size=1, alphabet=st.characters(whitelist_categories=("Ll",)))
rows = st.dictionaries(st.text(min_size=1), st.one_of(st.none(), st.text(), st.integers()))

@given(sources=st.lists(table_names, min_size=1, unique=True), row=rows, bryt=st.text(min_size=1))
def test_property_7_namespaced_data(sources, row, bryt):
    # Each source's columns are accessible under {table}.{column}.
    result = enrich(contract_data={"customerreference": bryt}, sources=sources, athena=lambda t: row)
    for table in sources:
        for column, value in row.items():
            assert result[table][column] == value

@given(row=rows, bryt=st.text(min_size=1))
def test_property_8_empty_pass_through(row, bryt):
    # No attachments -> unchanged data, no Athena calls.
    calls = []
    original = {"customerreference": bryt}
    result = enrich(contract_data=dict(original), sources=[], athena=lambda t: calls.append(t))
    assert result == original and calls == []

@given(sources=st.lists(table_names, min_size=1, unique=True), bryt=st.text(min_size=1))
def test_property_9_empty_rows_continue(sources, bryt):
    # Zero rows -> empty namespace, no raise.
    result = enrich(contract_data={"customerreference": bryt}, sources=sources, athena=lambda t: {})
    for table in sources:
        assert result.get(table, {}) == {}

@given(sources=st.lists(table_names, min_size=1, unique=True), bryt=st.text(min_size=1))
def test_property_10_failure_routes_to_handle_failure(sources, bryt):
    # Athena error -> the state throws (state machine routes to handle-failure).
    def boom(_): raise AthenaError("query failed")
    with pytest.raises(AthenaError):
        enrich(contract_data={"customerreference": bryt}, sources=sources, athena=boom)
```
