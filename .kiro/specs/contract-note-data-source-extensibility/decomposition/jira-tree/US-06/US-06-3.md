- **What:** Implement data source picker dialog (available sources excluding attached, with table/database/column count)
- **Why:** It is the [+ Attach] target opened by the Data Sources panel (US-06-2). It lets the Business_User pick from sources not yet attached and attaches the chosen one via `DataSourceService` (US-06-1).
- **Done when:**
  - The dialog lists available sources minus those already attached to the current template (`V \ A`), computed from `listAvailable` and the panel's attached set.
  - Each row shows table name, database name, and column count.
  - Selecting a source calls `DataSourceService.attach` and closes the dialog so the panel refreshes.
  - Attach validation errors (400 missing `bryt_number`, 404 table not found) surface in the dialog and it stays open.

### Suggested approach

```typescript
@Component({ selector: 'data-source-picker-dialog', templateUrl: './picker-dialog.html' })
export class DataSourcePickerDialogComponent implements OnInit {
  available: AvailableDataSource[] = [];

  constructor(
    private dataSources: DataSourceService,
    private ref: MatDialogRef<DataSourcePickerDialogComponent>,
    @Inject(MAT_DIALOG_DATA) private data: { templateId: string; attached: TemplateDataSource[] },
  ) {}

  ngOnInit(): void {
    this.dataSources.listAvailable().subscribe(all => {
      const attachedKeys = new Set(this.data.attached.map(a => `${a.database}.${a.tableName}`));
      // Property 12: present exactly V \ A.
      this.available = all.filter(v => !attachedKeys.has(`${v.database}.${v.tableName}`));
    });
  }

  select(ds: AvailableDataSource): void {
    this.dataSources.attach(this.data.templateId, ds.database, ds.tableName).subscribe({
      next: () => this.ref.close(ds),
      error: (err) => { /* surface 400/404 message; keep dialog open */ },
    });
  }
}
```

```html
<!-- picker-dialog.html -->
<table>
  <tr *ngFor="let ds of available" (click)="select(ds)">
    <td>{{ ds.tableName }}</td>
    <td>{{ ds.database }}</td>
    <td>{{ ds.columns.length }} columns</td>
  </tr>
</table>
```
