- **What:** Surface data source fields in the pdfme-designer palette for the edited variant: collapsible groups per data source; fields labelled {table}.{column} with type, visually distinct; placed fields use the namespaced name
- **Why:** Delivers `frontend-component:section-variant-field-browser` — the core capability that lets Business_Users design section variants with enriched data source data. Unblocks US-08 integration wiring.
- **Done when:**
  - The palette fetches the template's attached data sources and their columns via `DataSourceService` (list-attached, then get-columns per data source).
  - Each attached data source renders as one collapsible group; each column is a draggable field labelled `{table}.{column}` showing its column type, visually distinct (colour/icon) from core contract fields.
  - A placed field writes the namespaced `{table_name}.{column_name}` as the pdf-me element `name` into the variant's Schema_JSON.
  - For a shared section opened in a template context, the union of that template's attachments scopes the groups.
  - A failed get-columns / list-attached call surfaces a non-blocking palette error; core contract fields stay usable.

### Suggested approach

```typescript
// section-variant-field-browser: build palette groups for the edited variant.
// DataSourceService (US-06) wraps:
//   GET /contract-note-data-sources/{database}/{table}/columns
//   list-attached for the current template.

interface DataSourceColumn { name: string; type: string; }
interface TemplateDataSource { database: string; tableName: string; displayName: string; }

async buildDataSourceGroups(templateId: string): Promise<PaletteGroup[]> {
  const attached = await this.dataSourceService.listAttached(templateId);
  const groups: PaletteGroup[] = [];
  for (const ds of attached) {
    const columns = await this.dataSourceService.getColumns(ds.database, ds.tableName);
    groups.push({
      label: ds.displayName,
      collapsible: true,
      cssClass: 'palette-group--data-source', // visually distinct from core fields
      fields: columns.map((col: DataSourceColumn) => ({
        // namespaced name is what enrichment populates at render time
        name: `${ds.tableName}.${col.name}`,
        label: `${ds.tableName}.${col.name}`,
        type: col.type,
        draggable: true,
      })),
    });
  }
  return groups; // exactly the N attached data sources — no more, no fewer (Property 4)
}

// When a field is dropped on the pdfme-designer canvas, persist the namespaced name:
onFieldPlaced(field: PaletteField, schema: SchemaElement): void {
  schema.name = field.name; // e.g. "billing_accounts.account_number"
}
```
