- **What:** Display data source dependencies on the shared section detail screen
- **Why:** Delivers `frontend-component:shared-section-deps-display` — makes a shared section's tracked data source dependencies visible on its detail screen so Business_Users can see what a section requires before reusing it. Satisfies parent Requirement 4.5.
- **Done when:**
  - The shared section detail screen fetches dependencies via `DataSourceService.listSharedSectionDeps`.
  - Each dependency is listed with its database and table name.
  - A section with no dependencies shows an empty-state list rather than an error.

### Suggested approach

```typescript
// shared-section-deps-display: read-only list on the shared section detail screen.
// listSharedSectionDeps wraps:
//   GET /contract-note-shared-sections/{sharedSectionId}/data-source-dependencies

interface SectionDataSourceDependency { database: string; tableName: string; }

@Component({
  selector: 'shared-section-deps-display',
  template: `
    <section class="deps">
      <h3>Data source dependencies</h3>
      <ul *ngIf="deps.length; else none">
        <li *ngFor="let dep of deps">{{ dep.database }} · {{ dep.tableName }}</li>
      </ul>
      <ng-template #none><p class="empty">No data source dependencies.</p></ng-template>
    </section>
  `,
})
export class SharedSectionDepsDisplayComponent implements OnInit {
  @Input() sharedSectionId!: string;
  deps: SectionDataSourceDependency[] = [];

  constructor(private dataSourceService: DataSourceService) {}

  async ngOnInit(): Promise<void> {
    this.deps = await this.dataSourceService.listSharedSectionDeps(this.sharedSectionId);
  }
}
```
