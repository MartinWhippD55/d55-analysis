- **What:** Property tests for frontend logic (Property 4: field availability scoped to attachments; Property 6: missing dependency enforcement)
- **Why:** Pins the two invariants behind the field browser and the dependency check so refactors of either component keep their guarantees. Runs after tasks 1–3.
- **Done when:**
  - **Property 4** — for random attachment sets, the palette exposes fields from exactly the attached data sources (not more, not fewer).
  - **Property 6** — for random shared-section dependency sets and template attachments, the add flow prompts iff a required source is missing.
  - Both properties run in the frontend test suite (fast-check) and pass.

### Suggested approach

```typescript
import * as fc from 'fast-check';

const dsArb = fc.record({
  database: fc.constantFrom('cn_prod', 'cn_stage'),
  tableName: fc.string({ minLength: 1, maxLength: 8 }),
});
const key = (d: { database: string; tableName: string }) => `${d.database}#${d.tableName}`;

// Property 4: field availability scoped to attached data sources.
it('palette exposes exactly the attached data sources', () =>
  fc.assert(fc.property(fc.uniqueArray(dsArb, { selector: key }), (attached) => {
    const groups = buildDataSourceGroups(attached, columnsByTable);
    const exposed = new Set(groups.map(g => g.tableName));
    const expected = new Set(attached.map(d => d.tableName));
    expect(exposed).toEqual(expected); // no more, no fewer
  })));

// Property 6: missing dependency enforcement — prompt iff a required source is missing.
it('prompts iff a required data source is missing from the template', () =>
  fc.assert(fc.property(
    fc.uniqueArray(dsArb, { selector: key }), // deps
    fc.uniqueArray(dsArb, { selector: key }), // attachments
    (deps, attachments) => {
      const attachedKeys = new Set(attachments.map(key));
      const missing = deps.filter(d => !attachedKeys.has(key(d)));
      const decision = checkBeforeAttach(deps, attachments);
      expect(decision.canAttach).toBe(missing.length === 0);
    })));
```
