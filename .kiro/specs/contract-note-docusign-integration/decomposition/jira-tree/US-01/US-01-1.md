### Suggested approach

```typescript
// cdk/lib/contract-notes/docusign-pipeline.ts
import { Construct } from 'constructs';
import { RemovalPolicy } from 'aws-cdk-lib';
import * as dynamodb from 'aws-cdk-lib/aws-dynamodb';
import * as s3 from 'aws-cdk-lib/aws-s3';
import * as secretsmanager from 'aws-cdk-lib/aws-secretsmanager';

export interface DocuSignPipelineProps {
  resourcePrefix: string;
  errorOutputBucket: s3.IBucket; // reused from Estimate 1
}

export class DocuSignPipeline extends Construct {
  constructor(scope: Construct, id: string, props: DocuSignPipelineProps) {
    super(scope, id);
    const { resourcePrefix, errorOutputBucket } = props;

    const table = new dynamodb.Table(this, 'DocuSignEnvelopes', {
      tableName: `${resourcePrefix}docusign-envelopes`,
      partitionKey: { name: 'PK', type: dynamodb.AttributeType.STRING },
      sortKey: { name: 'SK', type: dynamodb.AttributeType.STRING },
      billingMode: dynamodb.BillingMode.PAY_PER_REQUEST,
    });
    table.addGlobalSecondaryIndex({
      indexName: 'SalesforceRefIndex',
      partitionKey: { name: 'GSI_PK', type: dynamodb.AttributeType.STRING },
      sortKey: { name: 'createdAt', type: dynamodb.AttributeType.STRING },
    });

    const signedBucket = new s3.Bucket(this, 'SignedContractNotes', {
      bucketName: `${resourcePrefix}signed-contract-notes`,
    });

    // Resource_Prefix-scoped credential secret placeholders
    new secretsmanager.Secret(this, 'DocuSignSecret', {
      secretName: `${resourcePrefix}contract-note/docusign`,
    });
    new secretsmanager.Secret(this, 'SalesforceSecret', {
      secretName: `${resourcePrefix}contract-note/salesforce`,
    });

    // Reuse Estimate 1's error bucket under the docusign/ prefix (no new bucket)
    errorOutputBucket.grantWrite(/* handler role */, 'docusign/*');

    // POST /docusign-webhook route surface (public for DocuSign_Connect);
    // route-to-Lambda binding + SendEnvelope task are wired in US-08.
    // NodejsFunction handlers from api/src/ attach later.
  }
}
```
