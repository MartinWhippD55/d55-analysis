- **What:** Shared TypeScript interfaces and types (EnvelopeRecord, EnvelopeStatus, ContractMetadata, SalesforceContact, CreateEnvelopeRequest, DocuSignWebhookEvent, error-record type) in shared-lib/src
- **Why:** Gives the `api` and `cdk` workspaces one consistent set of interfaces; every downstream handler (US-02 Salesforce client, US-03 DocuSign client, US-04 metadata service, US-05/US-06 Lambdas) consumes these types.
- **Done when:**
  - `EnvelopeRecord`, `EnvelopeStatus`, `ContractMetadata`, `SalesforceContact`, `CreateEnvelopeRequest`, `DocuSignWebhookEvent`, and the error-record type are defined in `shared-lib/src/`.
  - All of the above are exported from `shared-lib/src/index.ts` so both the `api` and `cdk` workspaces can import them.

### Suggested approach

```typescript
// shared-lib/src/index.ts
export type EnvelopeStatus =
  | 'sent' | 'delivered' | 'completed' | 'declined' | 'expired';

export interface EnvelopeRecord {
  envelopeId: string;
  salesforceRef: string;
  contractNoteS3Key: string;
  customerEmail: string;
  customerName: string;
  status: EnvelopeStatus;
  signedPdfS3Key?: string;
  createdAt: string;
  updatedAt: string;
  errorMessage?: string;
}

export interface ContractMetadata {
  salesforceRef: string;
  contractNoteS3Key: string;
  offerReference: string;
  customerName: string;
}

export interface SalesforceContact {
  contactId: string;
  firstName: string;
  lastName: string;
  email: string;
  accountId: string;
}

export interface CreateEnvelopeRequest {
  pdfBuffer: Buffer;
  recipientName: string;
  recipientEmail: string;
  documentName: string;
  emailSubject: string;
  webhookUrl: string;
}

export interface DocuSignWebhookEvent {
  event: string;
  apiVersion: string;
  data: { envelopeId: string; envelopeSummary?: unknown };
}

export interface ErrorRecord {
  timestamp: string;
  stage: string;
  envelopeId?: string;
  salesforceRef: string;
  contractNoteS3Key?: string;
  error: string;
  context: Record<string, unknown>;
}
```
