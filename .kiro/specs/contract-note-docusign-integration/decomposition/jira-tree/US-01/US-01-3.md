- **What:** Retry utility with exponential backoff (1s/2s/4s + jitter), configurable per-call
- **Why:** Reusable wrapper so external API calls survive transient failures; consumed by the DocuSign download (US-03) and Salesforce upload (US-02).
- **Done when:**
  - `withRetry(fn, opts)` retries a wrapped call up to a configurable max (default 3 attempts).
  - Backoff is exponential (1s/2s/4s) with jitter (±500ms) and is configurable per call.
  - The last error is re-thrown after the final attempt fails.

### Suggested approach

```typescript
// shared-lib/src/retry.ts
export interface RetryOptions {
  maxAttempts?: number; // default 3
  baseDelayMs?: number; // default 1000 -> 1s, 2s, 4s
  jitterMs?: number;    // default 500  -> ±500ms
}

export async function withRetry<T>(
  fn: () => Promise<T>,
  opts: RetryOptions = {},
): Promise<T> {
  const maxAttempts = opts.maxAttempts ?? 3;
  const baseDelayMs = opts.baseDelayMs ?? 1000;
  const jitterMs = opts.jitterMs ?? 500;

  let lastError: unknown;
  for (let attempt = 0; attempt < maxAttempts; attempt++) {
    try {
      return await fn();
    } catch (err) {
      lastError = err;
      if (attempt === maxAttempts - 1) break;
      const backoff = baseDelayMs * 2 ** attempt; // 1s, 2s, 4s
      const jitter = (Math.random() * 2 - 1) * jitterMs; // ±500ms
      await new Promise((r) => setTimeout(r, Math.max(0, backoff + jitter)));
    }
  }
  throw lastError;
}
```
