- **What:** Error record writer utility (JSON records to the reused error bucket)
- **Why:** Standard sink so all pipeline failures land in the reused error bucket in a consistent shape; used by the send (US-05) and webhook (US-06) handlers.
- **Done when:**
  - `writeErrorRecord(bucket, record)` serialises a JSON error record to `docusign/{timestamp}-{envelopeId}.json` in the reused `{resourcePrefix}contract-note-error-output` bucket.
  - The record contains timestamp, stage, envelopeId (if known), salesforceRef, error message, and context.
  - A failed S3 put is logged to CloudWatch (best-effort) rather than thrown.

### Suggested approach

```typescript
// shared-lib/src/error-writer.ts
import { PutObjectCommand, S3Client } from '@aws-sdk/client-s3';
import type { ErrorRecord } from './index';

const s3 = new S3Client({});

export async function writeErrorRecord(
  bucket: string,
  record: ErrorRecord,
): Promise<void> {
  const key = `docusign/${record.timestamp}-${record.envelopeId ?? 'unknown'}.json`;
  try {
    await s3.send(
      new PutObjectCommand({
        Bucket: bucket,
        Key: key,
        Body: JSON.stringify(record),
        ContentType: 'application/json',
      }),
    );
  } catch (err) {
    // Best-effort: the write is itself the failure sink.
    console.error('writeErrorRecord failed', { key, err });
  }
}
```
