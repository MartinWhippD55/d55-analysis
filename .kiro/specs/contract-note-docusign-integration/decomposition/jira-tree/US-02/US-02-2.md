- **What:** Customer contact lookup by customersalesforceref: return name + email; throw on not-found, missing email, or network error
- **Why:** The Send Envelope Lambda (US-05) resolves the signing recipient through this lookup; without a correct contact name + email, no envelope can be created for the right person.
- **Done when:**
  - `lookupContact(salesforceRef)` issues a `GET /services/data/v58.0/query` SOQL query keyed on `customersalesforceref` and returns a `SalesforceContact` (name + email).
  - It throws an error carrying the `customersalesforceref` on record-not-found and on network/credential errors.
  - It throws an error indicating a missing contact email when the record exists but has no email.

### Suggested approach

```typescript
import { SalesforceContact } from 'shared-lib/docusign-types'; // US-01

async function lookupContact(salesforceRef: string): Promise<SalesforceContact> {
  const token = await authenticate();
  const soql =
    `SELECT Id, FirstName, LastName, Email, AccountId FROM Contact ` +
    `WHERE customersalesforceref = '${salesforceRef}'`;

  let res: Response;
  try {
    res = await fetch(`${instanceUrl}/services/data/v58.0/query?q=${encodeURIComponent(soql)}`,
      { headers: { Authorization: `Bearer ${token}` } });
  } catch (err) {
    throw new Error(`Salesforce lookup failed for ${salesforceRef}: ${err}`);
  }

  const record = (await res.json()).records?.[0];
  if (!record) throw new Error(`Salesforce record not found for ${salesforceRef}`);
  if (!record.Email) throw new Error(`Missing contact email for ${salesforceRef}`);

  return {
    contactId: record.Id,
    firstName: record.FirstName,
    lastName: record.LastName,
    email: record.Email,
    accountId: record.AccountId,
  };
}
```
