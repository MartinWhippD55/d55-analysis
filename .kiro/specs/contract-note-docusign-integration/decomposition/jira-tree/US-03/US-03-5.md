- **What:** Property tests for the DocuSign client (Property 4: token management; Property 5: envelope contents; Property 6: HMAC validation)
- **Why:** Property-based tests give confidence that the client's core invariants hold across a wide range of inputs — token reuse/refresh, envelope contents, and HMAC validity. Optional and deferrable for a faster MVP.
- **Done when:**
  - Property 4 (token management): across any sequence of calls, a valid token is obtained and reused within its validity window, refreshing only when expired or near-expiry.
  - Property 5 (envelope contents): for any successfully created envelope, it contains the supplied PDF as the document and the supplied customer as the sole signer.
  - Property 6 (HMAC validation): `validateHmac` returns true only when the signature matches and false otherwise.

### Suggested approach

```typescript
import fc from 'fast-check';

// Property 5: envelope contains supplied document + sole signer
it('envelope contains correct document and recipient', () => {
  fc.assert(fc.property(
    fc.record({
      pdf: fc.uint8Array({ minLength: 1 }).map(Buffer.from),
      customerName: fc.string({ minLength: 1 }),
      customerEmail: fc.emailAddress(),
    }),
    (req) => {
      const env = buildEnvelopeDefinition(req as CreateEnvelopeRequest);
      expect(env.documents[0].documentBase64).toBe(req.pdf.toString('base64'));
      expect(env.recipients.signers).toHaveLength(1);
      expect(env.recipients.signers[0].email).toBe(req.customerEmail);
      expect(env.status).toBe('sent');
    },
  ));
});

// Property 6: HMAC valid only when signature matches
it('validateHmac accepts only matching signatures', () => {
  fc.assert(fc.property(
    fc.string(), fc.string({ minLength: 8 }),
    (body, hmacSecret) => {
      const good = require('crypto').createHmac('sha256', hmacSecret).update(body).digest('base64');
      expect(validateHmac(body, good, hmacSecret)).toBe(true);
      expect(validateHmac(body, good + 'x', hmacSecret)).toBe(false);
    },
  ));
});
```
