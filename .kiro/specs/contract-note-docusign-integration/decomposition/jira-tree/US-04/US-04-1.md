- **What:** DynamoDB metadata operations: create record on send, get by envelope ID, update status on webhook events, query by Salesforce_Ref via GSI
- **Why:** This is the core of `service:metadata-service`. It unblocks the send flow (US-05 calls `createRecord`) and the webhook flow (US-06 calls `getByEnvelopeId` then `updateStatus`), and provides the query-by-Salesforce-ref path for debugging. It also exposes the lookup by contract note S3 key that US-05 uses for its idempotency check — the idempotency decision itself lives in US-05.
- **Done when:**
  - `createRecord` puts an `EnvelopeRecord` with `PK = ENVELOPE#{envelopeId}`, `SK = METADATA`, and `createdAt`/`updatedAt` set.
  - `getByEnvelopeId` returns the record by base-table key, or `undefined` on a miss.
  - `updateStatus` sets the new `EnvelopeStatus`, refreshes `updatedAt`, and applies optional fields (e.g. `signedPdfS3Key`).
  - `queryBySalesforceRef` returns matching records via the `SalesforceRefIndex` GSI (`GSI_PK` = the Salesforce reference).

### Suggested approach

```typescript
import {
  DynamoDBDocumentClient,
  GetCommand,
  PutCommand,
  QueryCommand,
  UpdateCommand,
} from '@aws-sdk/lib-dynamodb';
import type { EnvelopeRecord, EnvelopeStatus } from 'shared-lib/docusign-types';

const TABLE = `${resourcePrefix}docusign-envelopes`;
const doc = DynamoDBDocumentClient.from(baseClient);

const key = (envelopeId: string) => ({ PK: `ENVELOPE#${envelopeId}`, SK: 'METADATA' });

export async function createRecord(record: EnvelopeRecord): Promise<void> {
  const now = new Date().toISOString();
  await doc.send(new PutCommand({
    TableName: TABLE,
    Item: { ...key(record.envelopeId), ...record, createdAt: now, updatedAt: now },
  }));
}

export async function getByEnvelopeId(envelopeId: string): Promise<EnvelopeRecord | undefined> {
  const { Item } = await doc.send(new GetCommand({ TableName: TABLE, Key: key(envelopeId) }));
  return Item as EnvelopeRecord | undefined;
}

export async function updateStatus(
  envelopeId: string,
  status: EnvelopeStatus,
  patch: { signedPdfS3Key?: string } = {},
): Promise<void> {
  await doc.send(new UpdateCommand({
    TableName: TABLE,
    Key: key(envelopeId),
    UpdateExpression: 'SET #s = :s, updatedAt = :u' +
      (patch.signedPdfS3Key ? ', signedPdfS3Key = :k' : ''),
    ExpressionAttributeNames: { '#s': 'status' },
    ExpressionAttributeValues: {
      ':s': status,
      ':u': new Date().toISOString(),
      ...(patch.signedPdfS3Key ? { ':k': patch.signedPdfS3Key } : {}),
    },
  }));
}

export async function queryBySalesforceRef(salesforceRef: string): Promise<EnvelopeRecord[]> {
  const { Items } = await doc.send(new QueryCommand({
    TableName: TABLE,
    IndexName: 'SalesforceRefIndex',
    KeyConditionExpression: 'GSI_PK = :ref',
    ExpressionAttributeValues: { ':ref': salesforceRef },
  }));
  return (Items ?? []) as EnvelopeRecord[];
}
```
