- **What:** Property tests for the metadata service (Property 8: record reflects current status)
- **Why:** Guards Property 8 — the stored record must reflect the current status. It protects US-06's webhook processing against ordering and overwrite bugs by exercising arbitrary sequences of status updates against the metadata service.
- **Done when:**
  - A fast-check property drives arbitrary sequences of `updateStatus` calls and asserts the record's `status` equals the last-applied `EnvelopeStatus`.
  - The property asserts `updatedAt` is monotonically newer after each applied update.
  - The test runs against DynamoDB local or a mocked `DynamoDBDocumentClient` and is marked optional (deferrable for a faster MVP).

### Suggested approach

```typescript
import fc from 'fast-check';
import type { EnvelopeStatus } from 'shared-lib/docusign-types';
import { createRecord, getByEnvelopeId, updateStatus } from './metadata-service';

const status = fc.constantFrom<EnvelopeStatus>(
  'sent', 'delivered', 'completed', 'declined', 'expired',
);

// Property 8: after any sequence of updates, the record reflects the last-applied status.
test('metadata record reflects current status', async () => {
  await fc.assert(
    fc.asyncProperty(fc.array(status, { minLength: 1 }), async (updates) => {
      const envelopeId = `env-${Math.random().toString(36).slice(2)}`;
      await createRecord({ envelopeId, status: 'sent' } as never);

      let prevUpdatedAt = (await getByEnvelopeId(envelopeId))!.updatedAt;
      for (const next of updates) {
        await updateStatus(envelopeId, next);
        const record = await getByEnvelopeId(envelopeId);
        expect(record!.status).toBe(next);
        expect(record!.updatedAt >= prevUpdatedAt).toBe(true);
        prevUpdatedAt = record!.updatedAt;
      }

      const final = await getByEnvelopeId(envelopeId);
      expect(final!.status).toBe(updates[updates.length - 1]);
    }),
  );
});
```
