- **What:** Send-envelope orchestration: Salesforce lookup -> DocuSign auth -> create envelope -> store metadata; on failure write error record + structured CloudWatch log
- **Why:** This is the core of the Lambda: it composes the US-02/US-03/US-04 services into the end-to-end send flow and captures failures for diagnosis. It runs after the US-05-1 validation + idempotency gate and produces the `EnvelopeRecord` that links the S3 key, Salesforce_Ref, and Envelope_ID.
- **Done when:**
  - The flow is orchestrated in order: `lookupContact(salesforceRef)` (US-02) → `authenticate()` + `createEnvelope(...)` (US-03) → `createRecord(...)` (US-04).
  - `createEnvelope` is passed the PDF from the render output location and the signing webhook URL; `createRecord` stores the envelope ID, Salesforce_Ref, contract note S3 key, customer email, name, and status `"sent"`.
  - On any stage failure, `writeErrorRecord(...)` (US-01) writes an error record to the error bucket and a structured JSON line is logged to CloudWatch including the S3 key, Salesforce_Ref, and Envelope_ID for that stage.
  - A metadata-write failure after a successful send is logged for manual reconciliation (envelope already sent).

### Suggested approach

```typescript
// api/src/docusign/send-envelope.ts (orchestration)
import { lookupContact } from "../salesforce/salesforce-client";
import { authenticate, createEnvelope } from "../docusign/docusign-client";
import { createRecord } from "../metadata/metadata-service";
import { writeErrorRecord } from "../shared/error-writer";
import type { ContractMetadata, EnvelopeRecord } from "../shared/docusign-types";

async function orchestrateSend(
  output: { bucket: string; key: string },
  metadata: ContractMetadata,
) {
  const contractNoteS3Key = output.key;
  let envelopeId: string | undefined;
  try {
    const contact = await lookupContact(metadata.salesforceRef);

    const auth = await authenticate();
    envelopeId = await createEnvelope(auth, {
      pdf: { bucket: output.bucket, key: output.key },
      recipient: contact,
      webhookUrl: process.env.SIGNING_WEBHOOK_URL,
    });

    const record: EnvelopeRecord = {
      envelopeId,
      salesforceRef: metadata.salesforceRef,
      contractNoteS3Key,
      email: contact.email,
      customerName: metadata.customerName,
      status: "sent",
    };
    await createRecord(record);
    log.info("envelope_sent", { contractNoteS3Key, salesforceRef: metadata.salesforceRef, envelopeId });
  } catch (err) {
    await writeErrorRecord({ contractNoteS3Key, salesforceRef: metadata.salesforceRef, envelopeId, error: err });
    log.error("send_envelope_failed", { contractNoteS3Key, salesforceRef: metadata.salesforceRef, envelopeId });
    throw err;
  }
}
```
