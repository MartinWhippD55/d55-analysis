- **What:** Specification tree validation utility
- **Why:** A single well-formedness check reused by the rules API (US-05) and the variant-rule API (US-04) so both reject malformed trees consistently. Depends on US-01-2's `SpecificationNode` type.
- **Done when:**
  - `shared-lib:spec-validation` confirms AND/OR nodes have both `leftOperand` and `rightOperand`, NOT nodes have their `operand`, and comparison leaves have a `field` plus `value` (EQUALS/LESS_THAN/MORE_THAN) or `values` (IN).
  - When the tree is incomplete, the validator returns errors that identify the incomplete nodes by path (rather than a single boolean).
  - When the tree is well-formed, the validator returns no errors.
  - It is a pure function with no I/O, importable by US-04 and US-05.

### Suggested approach

```typescript
import { SpecificationNode } from 'shared-lib/types';

export interface ValidationError {
  path: string;   // e.g. "root.rightOperand.leftOperand"
  message: string;
}

export function validateSpecification(
  node: SpecificationNode | undefined,
  path = 'root',
): ValidationError[] {
  if (!node) return [{ path, message: 'missing node' }];
  switch (node.type) {
    case 'AND':
    case 'OR': {
      const errors: ValidationError[] = [];
      if (!node.leftOperand) errors.push({ path: `${path}.leftOperand`, message: 'missing operand' });
      if (!node.rightOperand) errors.push({ path: `${path}.rightOperand`, message: 'missing operand' });
      return [
        ...errors,
        ...validateSpecification(node.leftOperand, `${path}.leftOperand`),
        ...validateSpecification(node.rightOperand, `${path}.rightOperand`),
      ];
    }
    case 'NOT':
      return node.operand
        ? validateSpecification(node.operand, `${path}.operand`)
        : [{ path: `${path}.operand`, message: 'missing operand' }];
    case 'IN':
      return node.field && Array.isArray(node.values) && node.values.length > 0
        ? []
        : [{ path, message: 'IN node requires field and values' }];
    default: // EQUALS / LESS_THAN / MORE_THAN
      return node.field && node.value !== undefined
        ? []
        : [{ path, message: 'comparison node requires field and value' }];
  }
}
```
