- **What:** Property tests for specification validation
- **Why:** Locks in the correctness guarantees the rules and variant-rule stories depend on, exercising the US-01-2 types and US-01-3 validator across generated trees rather than a few hand-picked cases. Optional for a faster MVP.
- **Done when:**
  - Property 20 (serialization round-trip): for any valid specification tree, serializing to JSON and deserializing produces an equivalent tree. Validates parent Requirements 10.2, 10.3, 10.4.
  - Property 21 (validation rejects malformed trees): for any structurally incomplete tree (AND/OR missing an operand, comparison missing field or value), validation fails and identifies the incomplete nodes. Validates parent Requirement 10.5.
  - Both properties run in the test suite with a generator that produces random valid and malformed `SpecificationNode` trees.

### Suggested approach

```typescript
import fc from 'fast-check';
import { validateSpecification } from 'shared-lib/spec-validation';
import { SpecificationNode } from 'shared-lib/types';

// Generator for well-formed specification trees (bounded depth).
const specArb: fc.Arbitrary<SpecificationNode> = fc.letrec((tie) => ({
  node: fc.oneof(
    { depthSize: 'small' },
    fc.record({ type: fc.constantFrom('EQUALS', 'LESS_THAN', 'MORE_THAN'),
                field: fc.string({ minLength: 1 }), value: fc.oneof(fc.string(), fc.integer()) }),
    fc.record({ type: fc.constant('IN'), field: fc.string({ minLength: 1 }),
                values: fc.array(fc.oneof(fc.string(), fc.integer()), { minLength: 1 }) }),
    fc.record({ type: fc.constant('NOT'), operand: tie('node') }),
    fc.record({ type: fc.constantFrom('AND', 'OR'),
                leftOperand: tie('node'), rightOperand: tie('node') }),
  ),
})).node as fc.Arbitrary<SpecificationNode>;

// Property 20: serialization round-trip.
test('spec tree survives JSON round-trip', () => {
  fc.assert(fc.property(specArb, (tree) => {
    expect(JSON.parse(JSON.stringify(tree))).toEqual(tree);
  }));
});

// Property 21: malformed trees are rejected with node paths.
test('validation rejects a tree with a stripped operand', () => {
  fc.assert(fc.property(specArb, (tree) => {
    const broken = { type: 'AND', leftOperand: tree } as unknown as SpecificationNode; // rightOperand missing
    const errors = validateSpecification(broken);
    expect(errors.length).toBeGreaterThan(0);
    expect(errors.some((e) => e.path.includes('rightOperand'))).toBe(true);
  }));
});
```
