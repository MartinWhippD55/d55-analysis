- **What:** get-linked-templates + publish-section-version handlers
- **Why:** These two handlers are the publish workflow — the mechanism that turns "every section edit goes live immediately" into an explicit, controlled rollout. `get-linked-templates` powers the confirmation list and update-available indicators; `publish-section-version` moves every linked template's pinned version. They unblock the render pipeline (US-06), which resolves the `pinnedVersionId` this handler maintains, and the frontend `SectionPublishComponent` (US-09/US-10).
- **Done when:**
  - `get-linked-templates` (`GET /contract-note-sections/{id}/linked-templates`) returns each linked template as a `SectionReference`: `templateId`, `templateName`, `pinnedVersionId`, `updateAvailable`.
  - `updateAvailable` is true if and only if the template's `pinnedVersionId` is older than the section's latest version (Property 33, parent 18.5).
  - `publish-section-version` (`POST /contract-note-sections/{id}/versions/{versionId}/publish`) sets `pinnedVersionId` to the chosen version on every `SHARED_SECTION#{sectionId}` / `REF#{templateId}` record; the chosen version defaults to the latest when not specified (Property 32, parent 18.4).
  - A Template Change Log entry (`TEMPLATE#{templateId}` / `CHANGELOG#{timestamp}`) is written for each affected template on publish.
  - Creating a section version (US-03) does not mutate any `pinnedVersionId` — publish is the only mutation (Property 31, parent 18.2).
  - A missing section or version returns a 404 not-found error; a DynamoDB write failure is logged and returns 500.

### Suggested approach

```typescript
// GET /contract-note-sections/{id}/linked-templates
// Item shape matches the SectionReference interface (shared-lib:types, US-01).
// (The response envelope below is a suggestion; the design pins the item shape,
//  not the wrapper.)
// {
//   "sectionId": "sec-123",
//   "latestVersionId": "v-2026-07-21T10-00-00",
//   "linkedTemplates": [
//     { "templateId": "tpl-1", "templateName": "Fixed CN",
//       "pinnedVersionId": "v-2026-07-20T09-00-00", "updateAvailable": true },
//     { "templateId": "tpl-2", "templateName": "Flex CN",
//       "pinnedVersionId": "v-2026-07-21T10-00-00", "updateAvailable": false }
//   ]
// }

export const getLinkedTemplates: Handler = async (event) => {
  const { id: sectionId } = event.pathParameters;
  const latestVersionId = await getLatestVersionId(sectionId); // from section-versions (US-03)
  // References live at PK = SHARED_SECTION#{sectionId}, SK begins_with REF#
  const refs = await queryReferences(sectionId);
  const linkedTemplates: SectionReference[] = refs.map((r) => ({
    templateId: r.templateId,
    templateName: r.templateName,
    pinnedVersionId: r.pinnedVersionId,
    updateAvailable: isOlder(r.pinnedVersionId, latestVersionId), // Property 33
  }));
  return ok({ sectionId, latestVersionId, linkedTemplates });
};

// POST /contract-note-sections/{id}/versions/{versionId}/publish
// versionId defaults to the section's latest version when omitted.
export const publishSectionVersion: Handler = async (event) => {
  const { id: sectionId, versionId } = event.pathParameters;
  const chosen = versionId ?? (await getLatestVersionId(sectionId)); // default to latest
  const refs = await queryReferences(sectionId);
  // Update pinnedVersionId on every REF# record + one change-log entry per template.
  await Promise.all(refs.flatMap((r) => [
    setPinnedVersion(sectionId, r.templateId, chosen), // Property 32
    writeChangeLog(r.templateId, {
      // NOTE: the design's Template Change Log changeType enum
      // (section-added/-removed/-reordered, metadata-updated, rule-updated)
      // does not list a publish value — see report (candidate spec gap).
      description: `Section ${sectionId} published at version ${chosen}`,
    }),
  ]));
  return ok({ sectionId, publishedVersionId: chosen, updatedTemplateCount: refs.length });
};
```
