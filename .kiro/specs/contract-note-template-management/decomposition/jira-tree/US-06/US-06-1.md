- **What:** Specification evaluator
- **Why:** `shared-lib:spec-evaluator` is the shared foundation reused by both template selection (US-06-2) and variant selection (US-06-4). Nothing in the pipeline can decide which template or variant applies until it exists, so it is a wave-1 task within this story.
- **Done when:**
  - A `shared-lib:spec-evaluator` function evaluates a `SpecificationNode` (from `shared-lib:types`, US-01) against a contract-data object and returns a boolean.
  - Leaf operators evaluate correctly: `EQUALS` (field === value), `IN` (value ∈ values), `LESS_THAN` and `MORE_THAN` (numeric comparison).
  - Logical operators `AND`, `OR`, `NOT` combine child results recursively.
  - Properties 22 and 23 hold (validated by the property tests in US-06-8).

### Suggested approach

```typescript
import type { SpecificationNode } from "shared-lib/types";

type ContractData = Record<string, unknown>;

export function evaluateSpec(node: SpecificationNode, data: ContractData): boolean {
  switch (node.operator) {
    case "EQUALS":
      return data[node.field] === node.value;
    case "IN":
      return node.values.includes(data[node.field] as never);
    case "LESS_THAN":
      return Number(data[node.field]) < Number(node.value);
    case "MORE_THAN":
      return Number(data[node.field]) > Number(node.value);
    case "AND":
      return node.children.every((child) => evaluateSpec(child, data));
    case "OR":
      return node.children.some((child) => evaluateSpec(child, data));
    case "NOT":
      return !evaluateSpec(node.child, data);
  }
}
```
