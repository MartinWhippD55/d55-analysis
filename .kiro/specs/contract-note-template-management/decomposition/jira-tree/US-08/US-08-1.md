- **What:** ContractNoteModule with routes + Cognito group route guard
- **Why:** Establishes the frontend module boundary for the contract-note admin area at `portal/src/app/components/contract-notes/`. It unblocks the US-09 screens, which render inside these routes, and gives US-10 the routes to link a sidebar entry to. It owns the story's one universal invariant: route access control (Property 39).
- **Done when:**
  - The `ContractNoteModule` exists at `portal/src/app/components/contract-notes/` and declares routes for the template list, template edit, shared sections and rules configuration areas.
  - A route guard is applied to those routes that grants access only to authenticated users holding the required Cognito group.
  - An unauthenticated user navigating to a module route is redirected to the login flow.
  - An authenticated user without the required Cognito group is shown an access-denied view.
  - Unit tests cover the guard: authenticated + in-group is allowed; unauthenticated is redirected; authenticated + not-in-group is denied.

### Suggested approach

```typescript
// portal/src/app/components/contract-notes/contract-note.routes.ts
import { Routes } from '@angular/router';
import { ContractNoteGroupGuard } from './contract-note-group.guard';

export const CONTRACT_NOTE_ROUTES: Routes = [
  {
    path: '',
    canActivate: [ContractNoteGroupGuard],
    children: [
      { path: '', redirectTo: 'templates', pathMatch: 'full' },
      { path: 'templates', component: TemplateListComponent },        // US-09
      { path: 'templates/:id', component: TemplateEditComponent },    // US-09
      { path: 'shared-sections', component: SharedSectionsComponent },// US-09
      { path: 'templates/:id/rules', component: RulesConfigComponent },// US-09
      { path: 'access-denied', component: AccessDeniedComponent },
    ],
  },
];

// contract-note-group.guard.ts
@Injectable({ providedIn: 'root' })
export class ContractNoteGroupGuard implements CanActivate {
  private readonly requiredGroup = 'contract-note-admins'; // TODO (spec gap): exact Cognito group name not stated in spec

  constructor(private auth: AuthService, private router: Router) {}

  canActivate(): boolean | UrlTree {
    if (!this.auth.isAuthenticated()) {
      return this.router.parseUrl('/login');           // parent 15.2
    }
    if (!this.auth.hasGroup(this.requiredGroup)) {
      return this.router.parseUrl('/contract-notes/access-denied'); // parent 15.3
    }
    return true;                                        // parent 15.1, Property 39
  }
}
```

Reuse the portal's existing auth mechanism (`AuthService`) rather than adding a new one.
