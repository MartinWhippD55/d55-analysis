- **What:** TemplateService, SectionService, RulesService wired to the API
- **Why:** Provides the data layer for the contract-note admin module. The US-09 screens inject these services to fetch and mutate real data; each service wraps the API Gateway endpoints delivered by US-02/US-03/US-04/US-05 and types payloads with the US-01 shared interfaces.
- **Done when:**
  - `TemplateService` provides `list`, `create`, `get`, `update`, `delete` and `reorder` over the `/contract-note-templates` endpoints.
  - `SectionService` provides section CRUD + reorder, schema get/save, version list/get/revert, shared-section CRUD + references, publish + linked-templates, and variant + variant-rule operations.
  - `RulesService` provides `get`/`save` of a template's specification over `/contract-note-templates/{id}/rule`, and is reusable for the variant-rule flow.
  - Every operation uses the correct HTTP method, URL and payload mapping, typed with the shared interfaces (`Template`, `Section`, `SharedSection`, `SectionVariant`, `SectionReference`, `SpecificationNode`).
  - Unit tests verify method/URL/payload for each operation using `HttpTestingController`.

### Suggested approach

```typescript
// portal/src/app/components/contract-notes/services/template.service.ts
@Injectable({ providedIn: 'root' })
export class TemplateService {
  private readonly base = `${environment.apiBaseUrl}/contract-note-templates`;
  constructor(private http: HttpClient) {}

  list(): Observable<Template[]> { return this.http.get<Template[]>(this.base); }
  create(body: Partial<Template>): Observable<Template> { return this.http.post<Template>(this.base, body); }
  get(id: string): Observable<Template> { return this.http.get<Template>(`${this.base}/${id}`); }
  update(id: string, body: Partial<Template>): Observable<Template> { return this.http.put<Template>(`${this.base}/${id}`, body); }
  delete(id: string): Observable<void> { return this.http.delete<void>(`${this.base}/${id}`); }
  reorder(order: string[]): Observable<void> { return this.http.put<void>(`${this.base}/reorder`, { order }); }
}

// section.service.ts (selected operations; base = `${apiBaseUrl}`)
listSections(templateId: string) { return this.http.get<Section[]>(`${b}/contract-note-templates/${templateId}/sections`); }
addSection(templateId: string, body: Partial<Section>) { return this.http.post<Section>(`${b}/contract-note-templates/${templateId}/sections`, body); }
removeSection(templateId: string, sectionId: string) { return this.http.delete<void>(`${b}/contract-note-templates/${templateId}/sections/${sectionId}`); }
reorderSections(templateId: string, order: string[]) { return this.http.put<void>(`${b}/contract-note-templates/${templateId}/sections/reorder`, { order }); }
getSchema(sectionId: string) { return this.http.get<object>(`${b}/contract-note-sections/${sectionId}/schema`); }
saveSchema(sectionId: string, schema: object) { return this.http.put<void>(`${b}/contract-note-sections/${sectionId}/schema`, schema); }
listVersions(sectionId: string) { return this.http.get<SectionVersion[]>(`${b}/contract-note-sections/${sectionId}/versions`); }
getVersion(sectionId: string, versionId: string) { return this.http.get<object>(`${b}/contract-note-sections/${sectionId}/versions/${versionId}`); }
revertVersion(sectionId: string, versionId: string) { return this.http.post<void>(`${b}/contract-note-sections/${sectionId}/versions/${versionId}/revert`, {}); }
getLinkedTemplates(sectionId: string) { return this.http.get<SectionReference[]>(`${b}/contract-note-sections/${sectionId}/linked-templates`); }
publishVersion(sectionId: string, versionId: string) { return this.http.post<void>(`${b}/contract-note-sections/${sectionId}/versions/${versionId}/publish`, {}); }
listVariants(sectionId: string) { return this.http.get<SectionVariant[]>(`${b}/contract-note-sections/${sectionId}/variants`); }
addVariant(sectionId: string, body: Partial<SectionVariant>) { return this.http.post<SectionVariant>(`${b}/contract-note-sections/${sectionId}/variants`, body); }
reorderVariants(sectionId: string, order: string[]) { return this.http.put<void>(`${b}/contract-note-sections/${sectionId}/variants/reorder`, { order }); }
updateVariant(sectionId: string, variantId: string, body: Partial<SectionVariant>) { return this.http.put<SectionVariant>(`${b}/contract-note-sections/${sectionId}/variants/${variantId}`, body); }
deleteVariant(sectionId: string, variantId: string) { return this.http.delete<void>(`${b}/contract-note-sections/${sectionId}/variants/${variantId}`); }
getVariantRule(sectionId: string, variantId: string) { return this.http.get<SpecificationNode>(`${b}/contract-note-sections/${sectionId}/variants/${variantId}/rule`); }
saveVariantRule(sectionId: string, variantId: string, spec: SpecificationNode) { return this.http.put<void>(`${b}/contract-note-sections/${sectionId}/variants/${variantId}/rule`, spec); }
listSharedSections() { return this.http.get<SharedSection[]>(`${b}/contract-note-sections/shared`); }
createSharedSection(body: Partial<SharedSection>) { return this.http.post<SharedSection>(`${b}/contract-note-sections/shared`, body); }
updateSharedSection(id: string, body: Partial<SharedSection>) { return this.http.put<SharedSection>(`${b}/contract-note-sections/shared/${id}`, body); }
deleteSharedSection(id: string) { return this.http.delete<void>(`${b}/contract-note-sections/shared/${id}`); }
getSharedSectionRefs(id: string) { return this.http.get<SectionReference[]>(`${b}/contract-note-sections/shared/${id}/references`); }

// rules.service.ts
getRule(templateId: string) { return this.http.get<SpecificationNode>(`${b}/contract-note-templates/${templateId}/rule`); }
saveRule(templateId: string, spec: SpecificationNode) { return this.http.put<void>(`${b}/contract-note-templates/${templateId}/rule`, spec); }
```

Payload/response shapes for `reorder` (`{ order: string[] }`) and the shared DTO field
names are illustrative — align exact request bodies with the US-02–US-05 handler contracts
and the US-01 shared types.
