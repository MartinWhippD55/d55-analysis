- **What:** TemplateEditComponent + change log + variant list
- **Why:** This is the hub screen reached from the template list. It composes the section list, the change log and the inline variants, and is where the section editor (US-09-4), version history (US-09-6) and rules editor (US-09-3) are launched. It is the parent of the wave-3 components, so it must land before them.
- **Done when:**
  - The name/description form is pre-populated on edit, enforces required fields, and surfaces a duplicate-name validation error from the API (parent 2.1, 2.3, 2.4, 3.1, 3.2).
  - The ordered section list renders in `sortOrder` order and supports add (new / shared / T&C), remove and reorder, persisted via `SectionService`; T&C sections stay positioned after all non-T&C sections (parent 3.3, 6.1, 6.2, 6.3, 6.4, 6.5).
  - Each section row shows a version badge and a history button that opens `SectionVersionHistoryComponent` (parent 17.2; delivered in US-09-6).
  - Sections with variants render the inline `SectionVariantsComponent` list on the page (parent 19.1, 21.4; delivered in US-09-5/variants work).
  - A collapsible change log panel lists the template's change events (timestamp, user, description) via `TemplateService` (parent 17.2, 17.3).
- **Backing endpoints:** `GET/PUT /contract-note-templates/{id}`, `GET/POST/DELETE /contract-note-templates/{id}/sections`, `PUT /contract-note-templates/{id}/sections/reorder`, template change-log read — all via `TemplateService` / `SectionService` (US-08).

### Suggested approach

```typescript
// template-edit.component.ts
@Component({ selector: 'app-template-edit', templateUrl: './template-edit.component.html' })
export class TemplateEditComponent implements OnInit {
  form = this.fb.group({
    name: ['', Validators.required],
    description: [''],
  });
  sections: Section[] = [];   // ordered by sortOrder
  changeLogOpen = false;
  changeLog: TemplateChangeLogEntry[] = [];

  constructor(
    private fb: FormBuilder,
    private templateService: TemplateService,
    private sectionService: SectionService,
    private route: ActivatedRoute,
    private dialog: DialogService,
  ) {}

  ngOnInit(): void {
    const id = this.route.snapshot.paramMap.get('id');
    if (id) {
      // GET /contract-note-templates/{id} -> Template (+ section list)
      this.templateService.get(id).subscribe((t) => this.form.patchValue(t));
      this.sectionService.list(id).subscribe((s) => (this.sections = s));
      this.templateService.changeLog(id).subscribe((c) => (this.changeLog = c));
    }
  }

  save(): void {
    if (this.form.invalid) return;               // required-field validation
    // PUT /contract-note-templates/{id}; 400 duplicate-name -> field error
    this.templateService.update(this.form.value).subscribe({
      error: (e) => this.applyServerErrors(e),   // e.g. "name already in use"
    });
  }

  openHistory(s: Section): void {
    this.dialog.open(SectionVersionHistoryComponent, { data: { sectionId: s.sectionId } });
  }

  reorderSections(order: string[]): void {
    // PUT /contract-note-templates/{id}/sections/reorder — revert on failure
    this.sectionService.reorder(this.templateId, order).subscribe();
  }
}
```

```html
<!-- template-edit.component.html (excerpt) -->
<form [formGroup]="form" (ngSubmit)="save()">
  <input formControlName="name" placeholder="Template name" />
  <textarea formControlName="description"></textarea>

<ul cdkDropList (cdkDropListDropped)="onDrop($event)">
  <li *ngFor="let s of sections" cdkDrag>
    {{ s.name }}
    <span class="version-badge">{{ s.pinnedVersionId | shortVersion }}</span>
    <button (click)="openHistory(s)">History</button>
    <app-section-variants *ngIf="s.hasVariants" [sectionId]="s.sectionId"></app-section-variants>
  </li>
</ul>

<button (click)="changeLogOpen = !changeLogOpen">Change log</button>
<div *ngIf="changeLogOpen">
  <div *ngFor="let c of changeLog">{{ c.createdAt }} — {{ c.createdBy }} — {{ c.description }}</div>
</div>
```
