- **What:** RulesConfigComponent (shared by template + variant rules)
- **Why:** This is the single visual specification editor reused in two places: template selection rules (from the template list / edit) and per-variant rules inside `SectionVariantsComponent`. Building it once and reusing it keeps rule editing and client-side validation consistent (parent 19.7). It carries the story's one correctness guarantee, Property 21.
- **Done when:**
  - The component renders a `SpecificationNode` tree with logical nodes AND/OR (binary, `leftOperand`/`rightOperand`), NOT (unary, `operand`) and leaf comparison nodes EQUALS, LESS_THAN, MORE_THAN, IN (parent 10.1, 10.2, 10.3).
  - Nodes can be added and removed, and leaf nodes let the user configure field, operator and value(s) (parent 10.1; IN takes a `values` array, the others a single `value`).
  - On save it serialises the tree to the specification JSON structure and runs a client-side well-formedness check that mirrors the server validation (parent 10.4, 10.5).
  - An incomplete tree (AND/OR missing an operand, or a comparison node missing field/value) blocks save and highlights the incomplete nodes (Property 21, parent 10.6).
  - The same component instance works bound to a template rule and to a variant rule (parent 19.7).
- **Backing endpoints:** template rules via `RulesService` — `GET/PUT /contract-note-templates/{id}/rule`; variant rules via `SectionService` — `GET/PUT /contract-note-sections/{id}/variants/{variantId}/rule` (US-08).

### Suggested approach

```typescript
// rules-config.component.ts — recursive tree editor; @Input the node, emit save.
type LogicalOperator = 'AND' | 'OR' | 'NOT';
type ComparisonOperator = 'EQUALS' | 'LESS_THAN' | 'MORE_THAN' | 'IN';

@Component({ selector: 'app-rules-config', templateUrl: './rules-config.component.html' })
export class RulesConfigComponent {
  @Input() node?: SpecificationNode;          // existing tree or undefined for new
  @Output() save = new EventEmitter<SpecificationNode>();

  incompleteNodeIds = new Set<string>();

  // Property 21 / parent 10.5, 10.6: block save + highlight incomplete nodes.
  validate(n: SpecificationNode | undefined, path = 'root'): boolean {
    if (!n) { this.incompleteNodeIds.add(path); return false; }
    switch (n.type) {
      case 'AND':
      case 'OR':
        return [
          this.validate(n.leftOperand, `${path}.left`),
          this.validate(n.rightOperand, `${path}.right`),
        ].every(Boolean);
      case 'NOT':
        return this.validate(n.operand, `${path}.operand`);
      case 'IN':
        return this.isSet(n.field) && Array.isArray(n.values) && n.values.length > 0;
      default: // EQUALS | LESS_THAN | MORE_THAN
        return this.isSet(n.field) && n.value !== undefined && n.value !== '';
    }
  }

  onSave(): void {
    this.incompleteNodeIds.clear();
    if (this.validate(this.node)) this.save.emit(this.node);
    // else: incompleteNodeIds drive the .invalid highlight in the template
  }

  private isSet(v?: string): boolean { return !!v && v.trim().length > 0; }
}
```

```html
<!-- rules-config.component.html — recursive self-reference for operands -->
<div class="node" [class.invalid]="incompleteNodeIds.has(path)">
  <select [(ngModel)]="node.type">
    <option>AND</option><option>OR</option><option>NOT</option>
    <option>EQUALS</option><option>LESS_THAN</option><option>MORE_THAN</option><option>IN</option>
  </select>

  <ng-container [ngSwitch]="node.type">
    <ng-container *ngSwitchCase="'AND'">
      <app-rules-config [node]="node.leftOperand"></app-rules-config>
      <app-rules-config [node]="node.rightOperand"></app-rules-config>
    </ng-container>
    <app-rules-config *ngSwitchCase="'NOT'" [node]="node.operand"></app-rules-config>
    <ng-container *ngSwitchDefault>
      <input [(ngModel)]="node.field" placeholder="field" />
      <!-- IN binds node.values (array); others bind node.value -->
      <input [(ngModel)]="node.value" placeholder="value" />
    </ng-container>
  </ng-container>
</div>
```
