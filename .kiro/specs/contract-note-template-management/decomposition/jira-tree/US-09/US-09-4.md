- **What:** SectionEditorComponent host (modal)
- **Why:** This is the Angular bridge to the US-07 `<pdfme-designer>` web component — the surface where a Business_User visually edits a section's layout. It is launched as a modal from the template edit page (US-09-2) and reused per-variant by the variants manager and read-only by version history preview (US-09-6).
- **Done when:**
  - The component presents `<pdfme-designer>` inside a modal (parent 7.1).
  - On open it loads the section's schema JSON from the API and passes it to the web component's `schema-json` attribute/property (parent 7.1, 7.3).
  - It listens for the designer's `schema-save` CustomEvent and persists the updated schema JSON via `SectionService` (parent 7.3).
  - If the designer fails to load, it renders an error state inside the modal with a retry button that re-attempts the load (parent 7.5).
  - Save and cancel actions close the modal; a save creates a new section version (server-side, US-03) rather than overwriting (parent 16.1).
- **Backing endpoints:** `GET /contract-note-sections/{id}/schema` (load), `PUT /contract-note-sections/{id}/schema` (save) — via `SectionService` (US-08). Hosts `web-component:pdfme-designer` (US-07).

### Suggested approach

```typescript
// section-editor.component.ts — modal host for <pdfme-designer> (US-07).
@Component({
  selector: 'app-section-editor',
  template: `
    <div class="modal">
      <ng-container *ngIf="!loadError; else errorState">
        <pdfme-designer #designer [attr.schema-json]="schemaJson"></pdfme-designer>
        <button (click)="close()">Cancel</button>
      </ng-container>
      <ng-template #errorState>
        <p>The section editor failed to load.</p>
        <button (click)="load()">Retry</button>
      </ng-template>
    </div>
  `,
  // custom element <pdfme-designer> is an unknown Angular element
  schemas: [CUSTOM_ELEMENTS_SCHEMA],
})
export class SectionEditorComponent implements AfterViewInit, OnDestroy {
  @Input() sectionId!: string;
  @ViewChild('designer') designerRef!: ElementRef<HTMLElement>;
  schemaJson = '';
  loadError = false;
  private onSave = (e: Event) => this.handleSave(e as CustomEvent);

  ngAfterViewInit(): void {
    this.designerRef.nativeElement.addEventListener('schema-save', this.onSave);
    this.load();
  }

  load(): void {
    this.loadError = false;
    // GET /contract-note-sections/{id}/schema -> { schemas: [...] }
    this.sectionService.getSchema(this.sectionId).subscribe({
      next: (schema) => (this.schemaJson = JSON.stringify(schema)),
      error: () => (this.loadError = true),      // parent 7.5 -> error state + retry
    });
  }

  private handleSave(e: CustomEvent): void {
    const schema = e.detail;                     // updated Schema_JSON from designer
    // PUT /contract-note-sections/{id}/schema -> creates a new version (US-03)
    this.sectionService.saveSchema(this.sectionId, schema).subscribe(() => this.close());
  }

  ngOnDestroy(): void {
    this.designerRef.nativeElement.removeEventListener('schema-save', this.onSave);
  }

  constructor(private sectionService: SectionService, private dialogRef: DialogRef) {}
  close(): void { this.dialogRef.close(); }
}
```
