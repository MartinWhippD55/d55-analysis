- **What:** SectionVersionHistoryComponent + SectionPublishComponent
- **Why:** These are the two surfaces that make section edits controllable. The version history modal (launched from a section row on the template edit page, US-09-2) lets a user review, preview and revert past versions; the publish component then pushes a chosen version out to the templates linked to that section. They front the US-04 publish/version API and are the last modals before navigation (US-09-7).
- **Done when:**
  - `SectionVersionHistoryComponent` lists versions with the current version highlighted and previous versions showing number, timestamp and user (parent 16.2).
  - A previous version can be previewed read-only in the section editor (parent 16.3).
  - Reverting a version creates a new version from the selected historical version (rather than deleting intermediate versions) after confirmation (parent 16.4).
  - The history surface can launch `SectionPublishComponent` for the section.
  - `SectionPublishComponent` lists the templates linked to the section, each with its pinned version and an update-available indicator (parent 18.3, 18.5).
  - Publishing a chosen version (defaulting to the latest) updates all linked templates after a confirmation and reports the templates updated (parent 18.4).
- **Backing endpoints:** `GET /contract-note-sections/{id}/versions`, `GET /contract-note-sections/{id}/versions/{versionId}`, `POST /contract-note-sections/{id}/versions/{versionId}/revert`, `GET /contract-note-sections/{id}/linked-templates`, `POST /contract-note-sections/{id}/versions/{versionId}/publish` — via `SectionService` (US-08, backed by US-04).

### Suggested approach

```typescript
// section-version-history.component.ts
@Component({ selector: 'app-section-version-history', templateUrl: './section-version-history.component.html' })
export class SectionVersionHistoryComponent implements OnInit {
  @Input() sectionId!: string;
  versions: SectionVersion[] = [];       // newest first; current flagged
  currentVersionId?: string;

  constructor(private sectionService: SectionService, private dialog: DialogService,
              private confirm: ConfirmDialogService) {}

  ngOnInit(): void {
    // GET /contract-note-sections/{id}/versions -> [{ versionId, createdAt, createdBy, ... }]
    this.sectionService.listVersions(this.sectionId).subscribe((v) => {
      this.versions = v;
      this.currentVersionId = v.find((x) => x.isCurrent)?.versionId;
    });
  }

  preview(v: SectionVersion): void {
    // open the section editor read-only on the historical version (parent 16.3)
    this.dialog.open(SectionEditorComponent,
      { data: { sectionId: this.sectionId, versionId: v.versionId, readOnly: true } });
  }

  async revert(v: SectionVersion): Promise<void> {
    if (!(await this.confirm.ask(`Revert to version ${v.versionId}? A new version will be created.`))) return;
    // POST .../versions/{versionId}/revert -> creates a new version (parent 16.4)
    this.sectionService.revert(this.sectionId, v.versionId).subscribe(() => this.ngOnInit());
  }

  publish(v: SectionVersion): void {
    this.dialog.open(SectionPublishComponent, { data: { sectionId: this.sectionId, versionId: v.versionId } });
  }
}

// section-publish.component.ts
@Component({ selector: 'app-section-publish', templateUrl: './section-publish.component.html' })
export class SectionPublishComponent implements OnInit {
  @Input() sectionId!: string;
  @Input() versionId?: string;           // defaults to latest when unset
  linked: SectionReference[] = [];       // { templateId, templateName, pinnedVersionId, updateAvailable }

  constructor(private sectionService: SectionService, private confirm: ConfirmDialogService) {}

  ngOnInit(): void {
    // GET /contract-note-sections/{id}/linked-templates -> SectionReference[]
    this.sectionService.linkedTemplates(this.sectionId).subscribe((l) => (this.linked = l));
  }

  async publish(): Promise<void> {
    if (!(await this.confirm.ask(`Publish to ${this.linked.length} linked template(s)?`))) return;
    // POST .../versions/{versionId}/publish (versionId defaults to latest server-side)
    this.sectionService.publish(this.sectionId, this.versionId).subscribe();
  }
}
```

```html
<!-- section-publish.component.html -->
<div *ngFor="let r of linked">
  {{ r.templateName }} — pinned {{ r.pinnedVersionId | shortVersion }}
  <span *ngIf="r.updateAvailable">update available</span>
</div>
<button (click)="publish()">Publish to all linked templates</button>
```
