  - `RulesConfigComponent` — tests cover tree add/remove manipulation and the validation display: a malformed tree blocks save and highlights the incomplete nodes (Property 21, parent 10).
  - `SectionEditorComponent` — tests cover the modal lifecycle, `schema-save` event handling, and the load-error/retry state (parent 7 support; requirement group 18/19 context).
  - `SectionVariantsComponent` — tests cover variant ordering and set-default logic (parent 19).
  - `SectionPublishComponent` — tests cover the publish-to-all confirmation flow and the update-available indicators (parent 18).

```typescript
// rules-config.component.spec.ts — Property 21 (parent 10.5, 10.6).
describe('RulesConfigComponent', () => {
  it('blocks save and highlights an incomplete AND node', () => {
    const fixture = TestBed.createComponent(RulesConfigComponent);
    const cmp = fixture.componentInstance;
    const saveSpy = jasmine.createSpy('save');
    cmp.save.subscribe(saveSpy);
    // AND node missing its rightOperand -> malformed tree
    cmp.node = { type: 'AND', leftOperand: { type: 'EQUALS', field: 'a', value: '1' } } as any;

    cmp.onSave();

    expect(saveSpy).not.toHaveBeenCalled();
    expect(cmp.incompleteNodeIds.has('root.right')).toBeTrue();
  });

  it('emits the serialised tree for a well-formed specification', () => {
    const fixture = TestBed.createComponent(RulesConfigComponent);
    const cmp = fixture.componentInstance;
    const saveSpy = jasmine.createSpy('save');
    cmp.save.subscribe(saveSpy);
    cmp.node = { type: 'EQUALS', field: 'offer.producttype', value: 'Fixed' } as any;

    cmp.onSave();

    expect(saveSpy).toHaveBeenCalledWith(cmp.node);
  });
});

// template-list.component.spec.ts — rendering + empty state (parent 1).
describe('TemplateListComponent', () => {
  it('shows the empty state when no templates exist', () => {
    const templateService = jasmine.createSpyObj('TemplateService', ['list']);
    templateService.list.and.returnValue(of([]));    // mocked US-08 service
    TestBed.configureTestingModule({
      declarations: [TemplateListComponent],
      providers: [{ provide: TemplateService, useValue: templateService }],
    });
    const fixture = TestBed.createComponent(TemplateListComponent);
    fixture.detectChanges();
    expect(fixture.nativeElement.textContent).toContain('No templates have been configured.');
  });
});
```
